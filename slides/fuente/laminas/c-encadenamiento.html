def compuerta(s: Estado) -> Literal["escribir_casos", "__end__"]:

    faltan = [p for p in ("Dado", "Cuando", "Entonces")

              if p not in s["criterios"]]

    return END if faltan else "escribir_casos"

 

grafo = StateGraph(Estado)

grafo.add_node("escribir_criterios", escribir_criterios)

grafo.add_node("escribir_casos", escribir_casos)

grafo.add_edge(START, "escribir_criterios")

grafo.add_conditional_edges("escribir_criterios", compuerta)

grafo.add_edge("escribir_casos", END)

app = grafo.compile()

Paralelización en código: abanico y unión

class Estado(TypedDict, total=False):

    historia: str

    casos: Annotated[list[str], operator.add] # reductor

 

def casos_api(s: Estado) -> Estado:

    return {"casos": [modelo().invoke("Casos de la API…").text]}

 

def casos_ui(s: Estado) -> Estado:

    return {"casos": [modelo().invoke("Casos de la UI…").text]}

 

grafo.add_edge(START, "casos_api") # abanico

grafo.add_edge(START, "casos_ui")

grafo.add_edge(["casos_api", "casos_ui"], "unir") # unión

grafo.add_edge("unir", END)

Lo que lo distingue

Dos aristas desde el mismo nodo, una unión con lista de origen y un reductor para el campo compartido.
